# Preserve raw command stdout without changing fixed padding or terminal newlines.
for path in (p/'validation').iterdir():
 if not path.is_file(): continue
 if path.name.endswith('.log'):
  (v/(path.name+'.gz')).write_bytes(gzip.compress(path.read_bytes(),mtime=0))
 elif path.name.endswith('.py'):
  shutil.copyfile(path,v/(path.name+'.txt'))
 elif path.name.endswith(('.xml','command.json')):
  shutil.copyfile(path,v/path.name)
