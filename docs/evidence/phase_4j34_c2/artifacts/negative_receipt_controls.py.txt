"""Re-signed software receipt controls; never scientific acceptance evidence."""
import json,shutil,subprocess,tempfile,os
from pathlib import Path
from island.forcefields.ambertools.runtime import signed,checksum
root=Path('/tmp/j34-c2-validation'); source=Path('docs/evidence/phase_4j34_c2')
env={**os.environ,'PYTHONPATH':'.:src','OPENMM_CPU_THREADS':'1','OMP_NUM_THREADS':'1','PYTHONDONTWRITEBYTECODE':'1'}
results=[]
for mutation in ('status','acceptance','readiness','runtime','downstream','counts','resigned_outcome'):
 with tempfile.TemporaryDirectory(prefix='c2-software-negative-') as tmp:
  target=Path(tmp)/'evidence';shutil.copytree(source,target)
  receipt=json.loads((target/'receipt.json').read_text())
  if mutation=='status':receipt['status']='complete'
  elif mutation=='acceptance':receipt['acceptance_met']=True
  elif mutation=='readiness':receipt['simulation_readiness']='established'
  elif mutation=='runtime':receipt['runtime_identity']='0'*64
  elif mutation=='downstream':receipt['downstream_gates']['npt']='passed'
  elif mutation=='counts':receipt['tests']['focused']['counts']['passed']+=1
  else:
   outcome=json.loads((target/'outcome.json').read_text());outcome['acceptance_met']=True
   outcome=signed({k:v for k,v in outcome.items() if k!='identity'})
   (target/'outcome.json').write_text(json.dumps(outcome,indent=2,sort_keys=True)+'\n')
   receipt['outcome_identity']=outcome['identity'];receipt['artifacts']['outcome.json']=checksum(target/'outcome.json')
  receipt=signed({k:v for k,v in receipt.items() if k!='identity'})
  (target/'receipt.json').write_text(json.dumps(receipt,indent=2,sort_keys=True)+'\n')
  argv=['../island-validation/phase4g1-env/bin/python','-m','scripts.validate_phase_4j34_c2_receipt',str(target/'receipt.json'),'--offline','--diagnostic-only']
  p=subprocess.run(argv,capture_output=True,text=True,env=env,check=False)
  assert p.returncode==1 and 'INVALID:' in p.stdout,(mutation,p.returncode,p.stdout,p.stderr)
  results.append({'software_negative_control':mutation,'argv':argv,'exit_code':p.returncode,'stdout':p.stdout,'stderr':p.stderr,'receipt_and_outcome_resigned':mutation=='resigned_outcome'})
(root/'receipt-negative-controls.json').write_text(json.dumps({'scientific_acceptance_evidence':False,'controls':results},indent=2)+'\n')
print('PASS',len(results),'re-signed software receipt negative controls')
