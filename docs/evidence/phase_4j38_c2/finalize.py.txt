import json,shutil,subprocess,hashlib,gzip,xml.etree.ElementTree as ET
from pathlib import Path
from island.analysis.npt import _signed
from scripts.validate_phase_4j38_c2_receipt import BASE,SOURCES,COMMANDS
implementation='0795329aed2c3bf75b7121d30aecfcd68bbc6bb0'

for name in ('run','validation'):
 shutil.copytree(r/name,out/name,dirs_exist_ok=True,ignore=shutil.ignore_patterns('*-tmp','__pycache__'))

 shutil.copyfile(r/name,out/name)
shutil.copyfile(r/'finalize.py',out/'finalize.py.txt')
shutil.copytree('/tmp/j38-c2-validation/before',out/'before-implementation',dirs_exist_ok=True)
with gzip.open(out/'before-implementation.log.gz','wb') as handle:handle.write(Path('/tmp/j38-c2-validation/before.log').read_bytes())
shutil.copyfile('/tmp/j38-c2-reproduce.py',out/'reproduce-before.py.txt')
write(out/'before-command.json',{'argv':['../island-validation/phase4g1-env/bin/python','/tmp/j38-c2-reproduce.py'],'environment':{'PYTHONPATH':'.:src','OPENMM_CPU_THREADS':'1','OMP_NUM_THREADS':'1'},'implementation_commit':BASE,'exit_code':0,'scope':'Unchanged C1 diagnostic reproduction, not scientific acceptance; source incompatibility remains blocked','script_sha256':sha('/tmp/j38-c2-reproduce.py')})

write(out/'base-verification.json',{'base_commit':BASE,'fetch_performed':True,'origin_branch':'codex/phase-4j38-c1-benchmark-charge-provenance','ls_remote_sha':subprocess.check_output(['git','ls-remote','origin','refs/heads/codex/phase-4j38-c1-benchmark-charge-provenance'],text=True).split()[0]})
tests={}
for name in ('focused','j22-j38-c1','full_supported'):
 tree=ET.parse(out/'validation'/(name+'.xml')).getroot()
 tests[name]={k:sum(int(s.get(k,0)) for s in tree.iter('testsuite')) for k in ('tests','failures','errors','skipped')}
full=ET.parse(out/'validation/full_supported.xml').getroot()
write(out/'skipped-tests.json',[{'name':t.get('name'),'classname':t.get('classname'),'reason':t.find('skipped').get('message')} for t in full.iter('testcase') if t.find('skipped') is not None])
manifest=_signed({'schema':'island.j38-c2-manifest.v1','base_commit':BASE,'tested_implementation_commit':implementation,'implementation_files':{p:sha(p) for p in sorted(SOURCES)},'files':{str(p.relative_to(out)):sha(p) for p in sorted(out.rglob('*')) if p.is_file() and p.name not in {'receipt.json','manifest.json'}}})
write(out/'manifest.json',manifest)
commands={name:json.loads((out/'validation'/(name+'-command.json')).read_text())['exit_code'] for name in sorted(COMMANDS)}
receipt=_signed({'schema':'island.j38-c2-receipt.v1','base_commit':BASE,'tested_implementation_commit':implementation,'manifest_identity':manifest['identity'],'matrix_identity':json.loads((out/'run/matrix.json').read_text())['identity'],'acceptance_met':False,'production_validated':False,'simulation_readiness':'not_established','new_md_executions':0,'new_md_executions_scope':'New C2 benchmark runs; regression suites execute established short integration runs','unmet_cases':['gaff2/aromatic','gaff2/pe','oplsaa/aromatic'],'outcome':'diagnostic_complete_acceptance_unmet','tests':tests,'validation_commands':commands,'historical_tracked_docs_and_evidence_files_preserved':len(json.loads((out/'historical-hashes.json').read_text())),'native_aromatic_classification':'source_incompatible','native_aromatic_component_residual_e':['-0.230','-0.230'],'native_aromatic_system_residual_e':'-0.460','formal_aromatic_system_total_e':0,'gaff2_blocker':'No independently reviewed raw Psi4/RESP geometry/ESP/native charge/executable artifacts for exact PE or aromatic final graph in declared inventory','charge_origins_kept_distinct':['native_oplsaa','external_provided_not_selected','RESP/Psi4','native_AM1BCC_blocked'],'charge_changes':False,'historical_evidence_rewritten':False,'final_evidence_commit_resolution':'git log -1 --format=%H -- docs/evidence/phase_4j38_c2/receipt.json'})
write(out/'receipt.json',receipt)
print(json.dumps({'tests':tests,'commands':commands,'receipt_identity':receipt['identity']},indent=2))
