"""Consistently re-signed software-only receipt controls; not scientific evidence."""
import json,shutil,subprocess,hashlib,os
from pathlib import Path
from island.analysis.npt import _signed
source=Path('docs/evidence/phase_4j38_c2');base=Path('/tmp/j38-c2-validation/publication-controls');base.mkdir(exist_ok=False)
env={**os.environ,'PYTHONPATH':'.:src','OPENMM_CPU_THREADS':'1','OMP_NUM_THREADS':'1'}
read=lambda p:json.loads(p.read_text())
def write(p,v):p.write_text(json.dumps(v,sort_keys=True,indent=2)+'\n')
def resign(v):return _signed({k:x for k,x in v.items() if k!='identity'})
results={}
for mutation in ('native_compatible','external_as_native','resp_other_graph','parameters_complete','receipt_acceptance','charge_vector'):
 target=base/mutation;shutil.copytree(source,target)
 receipt,manifest,matrix=read(target/'receipt.json'),read(target/'manifest.json'),read(target/'run/matrix.json')
 if mutation=='native_compatible':
  p=target/'run/native-source-decision.json';v=read(p);v['source_compatible']=True;write(p,resign(v))
 elif mutation=='external_as_native':matrix['cases']['oplsaa/aromatic']['selected_charge_origin']='external_provided'
 elif mutation=='resp_other_graph':matrix['cases']['gaff2/pe']['final_graph_identity']=matrix['cases']['gaff2/peo']['final_graph_identity']
 elif mutation=='parameters_complete':matrix['cases']['gaff2/aromatic']['stages']['parameter_completeness']['status']='passed'
 elif mutation=='receipt_acceptance':receipt['acceptance_met']=True
 else:
  p=target/'run/c1-recheck/aromatic-audit.json';v=read(p);v['rows'][0]['charge_e']+=.23;write(p,v)
 matrix=resign(matrix);write(target/'run/matrix.json',matrix);receipt['matrix_identity']=matrix['identity']
 manifest['files']={str(p.relative_to(target)):hashlib.sha256(p.read_bytes()).hexdigest() for p in target.rglob('*') if p.is_file() and p.name not in {'manifest.json','receipt.json'}}
 manifest=resign(manifest);write(target/'manifest.json',manifest);receipt['manifest_identity']=manifest['identity'];write(target/'receipt.json',resign(receipt))
 cmd=['../island-validation/phase4g1-env/bin/python','-m','scripts.validate_phase_4j38_c2_receipt',str(target/'receipt.json'),'--diagnostic-only']
 r=subprocess.run(cmd,env=env,capture_output=True,text=True,check=False)
 (base/(mutation+'.log')).write_text(r.stdout+r.stderr)
 assert r.returncode!=0,mutation
 results[mutation]={'argv':cmd,'exit_code':r.returncode,'classification':'software_negative_control','diagnostic_tail':(r.stdout+r.stderr).splitlines()[-1]}
 print(mutation,'rejected',flush=True)
write(base/'results.json',results)
