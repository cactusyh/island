import json,subprocess,os
from pathlib import Path
from scripts.validate_phase_4j35_c1 import signed
from island.forcefields.ambertools.resp import file_hash
root=Path('docs/evidence/phase_4j35_c1'); receipt=root/'receipt.json'
p=json.loads(receipt.read_text())
env={**os.environ,'PYTHONPATH':'.:src','OPENMM_CPU_THREADS':'1','OMP_NUM_THREADS':'1','PYTHONDONTWRITEBYTECODE':'1'}
for name,args,expected in [('corrective-diagnostic',['--diagnostic-only'],0),('corrective-acceptance',[],1)]:
 cmd=['../island-validation/phase4e2-env/bin/python','-m','scripts.validate_phase_4j35_c1_receipt',str(receipt),*args]
 run=subprocess.run(cmd,env=env,capture_output=True,text=True,check=False)
 assert run.returncode==expected and 'VALID BLOCKED DIAGNOSTIC' in run.stdout,(cmd,run.stdout,run.stderr)
 log=root/'artifacts'/(name+'.log');log.write_text(run.stdout+run.stderr)
 record={'argv':cmd,'implementation_commit':p['implementation_commit'],'exit_code':run.returncode,'expected_exit_code':expected,'scope':'diagnostic readability only; quantum acceptance remains unmet'}
 command=root/'artifacts'/(name+'-command.json');command.write_text(json.dumps(record,indent=2)+'\n')
 p['artifacts'][str(log.relative_to(root))]=file_hash(log)
 p['artifacts'][str(command.relative_to(root))]=file_hash(command)
 p['commands'][name]=record
receipt.write_text(json.dumps(signed({k:v for k,v in p.items() if k!='identity'}),indent=2)+'\n')
# Confirm the final, expanded receipt too; no self-referential receipt hash is
# written into its own validation log.
for args,expected in [(['--diagnostic-only'],0),([],1)]:
 run=subprocess.run(['../island-validation/phase4e2-env/bin/python','-m','scripts.validate_phase_4j35_c1_receipt',str(receipt),*args],env=env,capture_output=True,text=True,check=False)
 assert run.returncode==expected and 'VALID BLOCKED DIAGNOSTIC' in run.stdout,(run.stdout,run.stderr)
 print(run.returncode,run.stdout.strip())
