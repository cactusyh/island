import os,json,subprocess,time
from pathlib import Path
r=Path('/tmp/j34-validation');manifest=next((r/'amber-integration-tmp').rglob('references.json'))
env={**os.environ,'PYTHONPATH':'.:src','OPENMM_CPU_THREADS':'1','OMP_NUM_THREADS':'1','PYTHONDONTWRITEBYTECODE':'1','ISLAND_AMBERTOOLS_REFERENCE_MANIFEST':str(manifest)}
cmd=['../island-validation/phase4g1-env/bin/python','-m','pytest','-p','no:cacheprovider','-q','--tb=short','--junitxml='+str(r/'archived_reference.xml'),'--basetemp='+str(r/'archived-reference-tmp'),'tests/test_singlepoint_archived_references.py']
t=time.time()
with (r/'archived_reference.log').open('w') as log:p=subprocess.run(cmd,env=env,stdout=log,stderr=subprocess.STDOUT)
(r/'archived_reference-command.json').write_text(json.dumps({'argv':cmd,'environment':{k:env[k] for k in ('PYTHONPATH','OPENMM_CPU_THREADS','OMP_NUM_THREADS','PYTHONDONTWRITEBYTECODE','ISLAND_AMBERTOOLS_REFERENCE_MANIFEST')},'implementation_commit':subprocess.check_output(['git','rev-parse','HEAD'],text=True).strip(),'exit_code':p.returncode,'elapsed_seconds':time.time()-t},indent=2)+'\n')
print('archived_reference',p.returncode)
