import json, subprocess
from pathlib import Path
from island.forcefields.ambertools.psi4_runtime import environment, validate_runtime
r=json.loads(Path('docs/evidence/phase_4j36/environment.json').read_text());validate_runtime(r,expected_identity=r['identity'])
p=Path('/tmp/j39-validation/verified')
cmd=[r['prefix']+'/bin/python',str(Path('scripts/phase_4j37_refit.py').resolve()),'--directory',str(p/'run/raw/native-esp-aromatic'),'--output',str(p/'aromatic-native-refit.json')]
result=subprocess.run(cmd,env=environment(r['prefix']),capture_output=True,text=True,check=False,cwd='/tmp')
(p/'aromatic-native-refit.log').write_text(result.stdout+result.stderr)
(p/'aromatic-native-refit-command.json').write_text(json.dumps({'argv':cmd,'exit_code':result.returncode,'environment_policy':'pinned psi4_runtime.environment; 4 threads','implementation_commit':subprocess.check_output(['git','rev-parse','HEAD'],text=True).strip()},indent=2)+'\n')
assert result.returncode==0,(result.stdout,result.stderr)
print((p/'aromatic-native-refit.json').read_text())
